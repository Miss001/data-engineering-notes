select  
  a.table_name,
  (select sum(bytes)/1024/1024)/1024 from user_segments where segment_name=a.table_name and segment_type='TABLE') AS TABLE_SIZE_GB,
  (select sum(bytes)/1024/1024)/1024 from user_segments where segment_name in(select index_name from user_indexes where table_name=a.table_name) and segment_type='INDEX') as INDEX_SIZE_GB
from user_tables
