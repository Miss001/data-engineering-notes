# 创建表空间
```
create tablespace "TEST" datafile '/opt/dm8/data/PROD/TBS01.DBF' size 32 autoextend on next 2 maxsize 1024 CACHE = NORMAL;
```

# 扩展表空间
```
#扩展数据文件
Alter tablespace tbs resize datafile 'TBS01.DBF' to 100;
#添加数据文件
alter tablespace "TEST" add datafile '/opt/dm8/data/PROD/TBS02.DBF' size 32 autoextend on next 2 maxsize 1024 CACHE = NORMAL;
```
